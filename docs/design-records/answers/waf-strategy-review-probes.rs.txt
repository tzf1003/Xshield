    // ---- review probes (temporary, never committed) ----
    fn show(tag: &str, verdict: &Judgement, fake: &FakeJev) {
        let state = fake.sent.lock().unwrap().last().map(|s| state_of(s).to_owned()).unwrap_or_default();
        let cut: String = state.chars().take(900).collect();
        println!("PROBE {tag}: forward={} path={:?} outcome={:?} calls={} failure={:?}\n   state={}",
            verdict.forward, verdict.record.decision_path, verdict.record.outcome, fake.calls(),
            verdict.record.jev.as_ref().and_then(|j| j.failure.clone()), cut);
    }

    #[tokio::test]
    async fn probe_e1_fake_cookie() {
        let fake = FakeJev::new();
        let judge = judge(&fake);
        let site = site(r#","mode":"enforce""#);
        for path in ["/metrics", "/api/admin/export", "/users/alice", "/api/admin/deleteAll"] {
            let v = judge.judge(&site, &get(path, None, None, 1_000)).await;
            show(&format!("E1 no-cookie {path}"), &v, &fake);
            let v = judge.judge(&site, &get(path, None, Some("made-up"), 1_000)).await;
            show(&format!("E1 fake-cookie {path}"), &v, &fake);
        }
        let v = judge.judge(&site, &RequestFacts { method: "DELETE", ..get("/api/patients", None, Some("made-up"), 1_000) }).await;
        show("E1 DELETE collection", &v, &fake);
        let v = judge.judge(&site, &RequestFacts { method: "OPTIONS", ..get("/api/patients/7", None, None, 1_000) }).await;
        show("E7 OPTIONS preflight", &v, &fake);
    }

    #[tokio::test]
    async fn probe_e4_multipart_and_long() {
        let fake = FakeJev::new();
        let judge = judge(&fake);
        let site = site(r#","mode":"enforce""#);
        judge.observe(&site, &list_response("alice", 0));
        let filler = "A".repeat(600);
        let body = format!("--b\r\nContent-Disposition: form-data; name=\"note\"\r\n\r\n{filler}\r\n--b\r\nContent-Disposition: form-data; name=\"patient_id\"\r\n\r\n8\r\n--b\r\nContent-Disposition: form-data; name=\"q\"\r\n\r\n' OR 1=1--\r\n--b--\r\n");
        let v = judge.judge(&site, &RequestFacts {
            method: "POST", content_type: Some("multipart/form-data; boundary=b"),
            body: body.as_bytes(),
            ..get("/api/records", None, Some("alice"), 1_000) }).await;
        show("E4 multipart", &v, &fake);
        let sent = fake.sent.lock().unwrap().last().cloned().unwrap();
        println!("   multipart: state mentions patient_id={} sqli={}", sent.contains("patient_id"), state_of(&sent).contains("OR 1=1"));
        let long = format!("{}%27%20OR%201%3D1--", "A".repeat(600));
        let q = format!("comment={long}");
        let v = judge.judge(&site, &get("/api/records", Some(&q), Some("alice"), 1_000)).await;
        show("E5 long value", &v, &fake);
        let sent = fake.sent.lock().unwrap().last().cloned().unwrap();
        println!("   long: state has sqli={}", state_of(&sent).contains("OR 1=1"));
        // text/plain JSON
        let v = judge.judge(&site, &RequestFacts {
            method: "POST", content_type: Some("text/plain"),
            body: br#"{"patient_id":8}"#,
            ..get("/api/records", None, Some("alice"), 1_000) }).await;
        show("E4 text/plain json", &v, &fake);
    }

    #[tokio::test]
    async fn probe_e14_shared_address() {
        let fake = FakeJev::new();
        let judge = judge(&fake);
        let site = SiteSpec::from_json(r#"{"name":"emr","hosts":["emr.test"],"origin":"http://o:1","session":{"cookie":"SID"},
               "entries":["/login","/api/users","/api/security-answers"],"mode":"enforce"}"#).unwrap();
        // user 1 signs up from the load balancer's address; the origin shows the new user's id
        judge.observe(&site, &Observation { credential: None, client: "10.0.0.9", method: "POST", path: "/api/users", body: br#"{"id":"4711","email":"u1@x.cc"}"#, ..list_response("x", 0) });
        // another client behind the same address names that user id on an entry
        let v = judge.judge(&site, &RequestFacts { method: "POST", content_type: Some("application/json"), body: br#"{"UserId":"4711","answer":"x"}"#, ..get("/api/security-answers", None, None, 1_000) }).await;
        show("E14 other client same address", &v, &fake);
    }

    #[tokio::test]
    async fn probe_e15_search_by_id_then_open() {
        let fake = FakeJev::new();
        let judge = judge(&fake);
        let site = site(r#","mode":"enforce""#);
        // the user types an order number into the search box; Jev allows the search input
        let search = get("/api/orders/search", Some("q=20261113001"), Some("alice"), 1_000);
        let v = judge.judge(&site, &search).await;
        show2("E15 search", &v, &fake);
        let n = judge.observe(&site, &Observation { path: "/api/orders/search", body: br#"{"results":[{"id":"20261113001","title":"mine"}]}"#, sent: &v.sent, ..list_response("alice", 1_100) });
        println!("   search response taught {} new values (of {})", n.new_values, n.values);
        fake.answer("DENY", "BENIGN");
        let v = judge.judge(&site, &get("/api/orders/20261113001", None, Some("alice"), 2_000)).await;
        show2("E15 open the result", &v, &fake);
        // later the session's own order list shows the same order
        let n = judge.observe(&site, &Observation { path: "/api/orders", body: br#"{"orders":[{"id":"20261113001"}]}"#, ..list_response("alice", 3_000) });
        println!("   own list taught {} new values (of {})", n.new_values, n.values);
        let v = judge.judge(&site, &get("/api/orders/20261113001", None, Some("alice"), 4_000)).await;
        show2("E15 open after own list", &v, &fake);
    }

    #[tokio::test]
    async fn probe_e17_fifo() {
        let fake = FakeJev::new();
        let judge = judge(&fake);
        let site = site(r#","mode":"enforce""#);
        judge.observe(&site, &list_response("alice", 0)); // shows patient 7
        for page in 0..3u32 {
            let mut body = String::from("{\"items\":[");
            for i in 0..1900u32 { if i > 0 { body.push(','); } body.push_str(&format!("{{\"id\":\"X{}\"}}", page * 10_000 + i)); }
            // every page shows patient 7 again
            body.push_str(",{\"id\":\"7\"}]}");
            let leaked: &'static [u8] = Box::leak(body.into_bytes().into_boxed_slice());
            judge.observe(&site, &Observation { path: "/api/patients", body: leaked, ..list_response("alice", 1_000 + u64::from(page)) });
        }
        fake.answer("DENY", "BENIGN");
        let v = judge.judge(&site, &get("/api/patients/7", None, Some("alice"), 5_000)).await;
        show2("E17 patient 7 re-shown on every page, after 3 big pages", &v, &fake);
    }
