    // 规格测试 3 的“不暂存登录”：JSON 凭证站点，登录响应正文里带令牌、没有 Set-Cookie。
    #[tokio::test]
    async fn probe_refused_json_login_is_not_held() {
        let fake = FakeJev::new();
        fake.answer("DENY", "BENIGN");
        let judge = judge(&fake);
        let site = site(r#","mode":"observe","advanced":{"field_kinds":[{"field":"bid","kind":"basket"}]}"#);
        let body = login_body(LOGIN_TOKEN, 6);
        let login = judge.observe(
            &site,
            &Observation {
                would_block: true,
                ..json_login(body.as_bytes(), "10.0.0.1", 0)
            },
        );
        assert_eq!(login.pending_values, None);
        assert_eq!(login.skip, Some(Skip::WouldBlock));
        assert_eq!(login.new_values, 0);
        let entry = judge
            .judge(&site, &get("/api/home", None, Some(LOGIN_TOKEN), 1))
            .await;
        // 这里应断言 entry.record.trace.session 的 login_request_id 为 None。
        let own = judge
            .judge(&site, &get("/rest/basket/6", None, Some(LOGIN_TOKEN), 2))
            .await;
        assert_ne!(own.record.decision_path, DecisionPath::FastAllow);
        // 地址上下文也没有学
        let anon = judge
            .judge(
                &site,
                &RequestFacts {
                    client: "10.0.0.1",
                    ..get("/api/patients", Some("id=6"), None, 3)
                },
            )
            .await;
        assert_ne!(anon.record.decision_path, DecisionPath::FastAllow);
        let _ = entry;
    }

    // 已有会话：不记入口通过、不上报结构和指纹、不算属性越权告警。
    #[tokio::test]
    async fn probe_refused_response_keeps_no_trace_in_a_live_session() {
        let fake = FakeJev::new();
        fake.answer("DENY", "BENIGN");
        let judge = judge(&fake);
        let site = site(r#","mode":"observe""#);
        confirm(&judge, &site, "alice");
        let observed = judge.observe(
            &site,
            &Observation {
                would_block: true,
                path: "/api/home",
                ..list_response("alice", 1_000)
            },
        );
        assert!(!observed.entry_passed);
        assert_eq!(observed.new_values, 0);
        assert_eq!(observed.values, 0);
        assert!(observed.fingerprints.is_empty());
        assert!(observed.shape.is_none());
        assert!(observed.exposure.is_none());
        // 同一响应不带标记时才第一次经过入口：说明前一次什么都没记。
        let again = judge.observe(
            &site,
            &Observation {
                path: "/api/home",
                ..list_response("alice", 2_000)
            },
        );
        assert!(again.entry_passed);
        assert!(again.new_values > 0);
    }

    // 判断顺序：签发凭证的重定向和非 2xx 仍记 status，绕过路径仍记 bypass。
    #[tokio::test]
    async fn probe_refused_redirect_and_bypass_keep_their_reasons() {
        let fake = FakeJev::new();
        let judge = judge(&fake);
        let site = site(r#","mode":"observe""#);
        let redirect = judge.observe(
            &site,
            &Observation {
                would_block: true,
                credential: None,
                issued_credential: Some("fresh"),
                status: 302,
                path: "/login",
                ..list_response("x", 1_000)
            },
        );
        assert_eq!(redirect.skip, Some(Skip::Status));
        assert!(redirect.issued_session.is_some());
        let bypass = judge.observe(
            &site,
            &Observation {
                would_block: true,
                path: "/static/app.js",
                ..list_response("alice", 1_000)
            },
        );
        assert_eq!(bypass.skip, Some(Skip::Bypass));
        let failed = judge.observe(
            &site,
            &Observation {
                would_block: true,
                status: 500,
                ..list_response("alice", 1_000)
            },
        );
        assert_eq!(failed.skip, Some(Skip::Status));
    }
